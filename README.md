                  <div className="absolute left-1/2 top-0 bottom-0 w-px bg-gray-300 z-10"></div>
                  <div className={`h-full ${trait.color} transition-all duration-1000 ease-out`} style={{ width: `${fillWidth}%` }}></div>
                  <div className="absolute right-3 font-bold text-sm text-[#2C2425]/50 md:hidden z-20 mix-blend-overlay">{score}</div>
                </div>
              </div>
            );
          })}
        </div>
      </div>

      <div className="text-center pb-10">
         <button onClick={() => { setCurrentScreen('level2_intro'); window.scrollTo({ top: 0, behavior: 'smooth' }); }} className="bg-gradient-to-r from-purple-500 to-indigo-500 hover:from-purple-600 hover:to-indigo-600 text-white px-10 py-4 rounded-full font-medium text-lg transition-all hover:scale-105 active:scale-95 shadow-lg shadow-indigo-200/50 inline-flex items-center gap-2">
            {t.resConsult} <ArrowRightIcon />
          </button>
      </div>
    </div>
  );

  const renderLevel2Intro = () => (
    <div className="max-w-3xl mx-auto bg-white/90 backdrop-blur-md rounded-[2.5rem] p-8 md:p-12 shadow-[0_8px_40px_rgba(99,102,241,0.08)] border border-indigo-50 animate-fade-in mt-10">
      <div className="mb-10 text-center">
        <h2 className="text-2xl md:text-4xl font-bold text-[#2C2425] mb-6 leading-tight">{t.lvl2Title}</h2>
        
        <div className="bg-indigo-50 rounded-3xl p-6 mb-8 inline-block shadow-inner border border-indigo-100">
          <div className="grid grid-cols-2 md:grid-cols-4 gap-6 text-center">
            <div><div className="text-sm text-indigo-400 font-semibold mb-1">День</div><div className="text-3xl font-bold text-indigo-600">{parsedDob?.day}</div></div>
            <div><div className="text-sm text-indigo-400 font-semibold mb-1">Месяц</div><div className="text-3xl font-bold text-indigo-600">{parsedDob?.month}</div></div>
            <div><div className="text-sm text-indigo-400 font-semibold mb-1">Год</div><div className="text-3xl font-bold text-indigo-600">{parsedDob?.lastDigit}</div></div>
            <div><div className="text-sm text-indigo-400 font-semibold mb-1">Код</div><div className="text-3xl font-bold text-indigo-600">{parsedDob?.code}</div></div>
          </div>
        </div>

        <p className="text-lg text-[#6B5A5C] leading-relaxed max-w-2xl mx-auto">{t.lvl2IntroText}</p>
      </div>

      <div className="text-center pt-6">
        <button onClick={startLevel2} className="bg-gradient-to-r from-purple-500 to-indigo-500 hover:from-purple-600 hover:to-indigo-600 text-white px-10 py-4 rounded-full font-medium text-lg transition-all hover:scale-105 active:scale-95 shadow-lg shadow-indigo-200/50 inline-flex items-center gap-2">
          {t.btnStartLvl2} <ArrowRightIcon />
        </button>
      </div>
    </div>
  );

  const renderResult2 = () => {
    const dayDesc = getDesc(descDay, parsedDob?.day);
    const monthDesc = getDesc(descMonth, parsedDob?.month);
    const yearDesc = getDesc(descYear, parsedDob?.lastDigit);
    const codeDesc = getDesc(descCode, parsedDob?.code);

    const config2 = [
      { id: '1', header: 'БЛОК 1: ДЕНЬ РОЖДЕНИЯ', subheader: '(главное предназначение в личной сфере, фундамент характера, таланты, потенциал)', desc: dayDesc },
      { id: '2', header: 'БЛОК 2: МЕСЯЦ РОЖДЕНИЯ', subheader: '(духовное предназначение, связь, наши скрытые таланты, какие уроки нам нужно пройти для роста)', desc: monthDesc },
      { id: '3', header: 'БЛОК 3: ПОСЛЕДНЯЯ ЦИФРА ГОДА РОЖДЕНИЯ', subheader: '(точка опоры, ключевой навык, главная черта характера)', desc: yearDesc },
      { id: '4', header: 'БЛОК 4: ПЕРВЫЙ КОД СУДЬБЫ', subheader: '(как мозг воспринимает реальность и реагирует на нее)', desc: codeDesc },
    ];

    return (
      <div className="max-w-4xl mx-auto animate-fade-in mt-10">
        <div className="text-center mb-12">
          <div className="inline-flex items-center justify-center w-20 h-20 bg-indigo-100 text-indigo-600 rounded-full mb-6"><CheckIcon /></div>
          <h2 className="text-3xl md:text-5xl font-bold text-[#2C2425] mb-4">{t.res2Title}</h2>
          <p className="text-xl text-[#6B5A5C]">{t.res2Subtitle}</p>
        </div>

        <div className="space-y-12 mb-10">
          {config2.map((block, idx) => {
            const scores = mockResults2[idx] || { plusPercent: 0, minusPercent: 0 };
            const isPlusDominant = scores.plusPercent >= scores.minusPercent;

            return (
              <div key={block.id} className="bg-white rounded-[2rem] p-6 md:p-10 shadow-[0_8px_40px_rgba(99,102,241,0.08)] border border-indigo-50">
                <div className="mb-8">
                  <h3 className={`text-xl md:text-2xl font-bold uppercase tracking-wide bg-clip-text text-transparent ${isPlusDominant ? 'bg-gradient-to-r from-green-600 to-emerald-400' : 'bg-gradient-to-r from-rose-600 to-orange-400'}`}>
                    {block.header}
                  </h3>
                  <p className="text-sm md:text-base text-gray-500 mt-2 font-medium">{block.subheader}</p>
                  <div className="mt-4 inline-block bg-indigo-50 text-indigo-800 px-4 py-1.5 rounded-full font-bold text-sm md:text-base border border-indigo-100">
                    {block.desc.title}
                  </div>
                </div>

                <div className="space-y-6">
                  {/* ШКАЛА 1: ПЛЮСОВАЯ ЭНЕРГИЯ */}
                  <div className="relative bg-gray-100 rounded-2xl overflow-hidden min-h-[140px] flex flex-col justify-center border border-gray-200">

                    <div className="relative z-10 p-5 md:p-6 flex flex-col md:flex-row gap-4 md:items-center">
                      <div className="flex-shrink-0 w-16 h-16 bg-white/80 backdrop-blur-sm rounded-2xl flex flex-col items-center justify-center shadow-sm border border-green-200 text-green-700">
                        <span className="text-xl font-black">{scores.plusPercent}%</span>
                      </div>
                      <div>
                        <h4 className="font-bold text-green-900 text-lg mb-1 flex items-center gap-2">
                          Плюс (+) Энергия
                          {isPlusDominant && <span className="text-xs bg-green-500 text-white px-2 py-0.5 rounded-full uppercase tracking-wider">Доминирует</span>}
                        </h4>
                        <p className="text-sm md:text-base text-green-950/80 leading-relaxed font-medium">{block.desc.plus}</p>
                      </div>
                    </div>
                  </div>

                  {/* ШКАЛА 2: МИНУСОВАЯ ЭНЕРГИЯ */}
                  <div className="relative bg-gray-100 rounded-2xl overflow-hidden min-h-[140px] flex flex-col justify-center border border-gray-200">
                    <div className="absolute left-0 top-0 bottom-0 bg-gradient-to-r from-rose-100 to-orange-200 transition-all duration-1000 ease-out z-0" style={{ width: `${scores.minusPercent}%` }}></div>
                    <div className="relative z-10 p-5 md:p-6 flex flex-col md:flex-row gap-4 md:items-center">
                      <div className="flex-shrink-0 w-16 h-16 bg-white/80 backdrop-blur-sm rounded-2xl flex flex-col items-center justify-center shadow-sm border border-rose-200 text-rose-700">
                        <span className="text-xl font-black">{scores.minusPercent}%</span>
                      </div>
                      <div>
                        <h4 className="font-bold text-rose-900 text-lg mb-1 flex items-center gap-2">
                          Минус (-) Энергия
                          {!isPlusDominant && <span className="text-xs bg-rose-500 text-white px-2 py-0.5 rounded-full uppercase tracking-wider">Доминирует</span>}
                        </h4>
                        <p className="text-sm md:text-base text-rose-950/80 leading-relaxed font-medium">{block.desc.minus}</p>
                      </div>
                    </div>
                  </div>
                </div>
              </div>
            );
          })}
        </div>

        <div className="text-center pb-10">
           <button className="bg-[#1D1D1F] hover:bg-black text-white px-10 py-4 rounded-full font-medium text-lg transition-all hover:scale-105 active:scale-95 shadow-lg inline-flex items-center gap-2">
              {t.res2Consult} <ArrowRightIcon />
            </button>
        </div>
      </div>
    );
  };

  return (
    <div className="relative min-h-screen bg-[#FCFAFA] text-[#2C2425] font-sans selection:bg-[#D37B85] selection:text-white pb-24 overflow-hidden">
      <div className={`absolute top-[-10%] left-[-10%] w-[40rem] h-[40rem] ${currentScreen.includes('2') ? 'bg-indigo-200/40' : 'bg-rose-200/40'} rounded-full mix-blend-multiply filter blur-[120px] opacity-70 transition-colors duration-1000`}></div>
      <div className={`absolute top-[20%] right-[-5%] w-[35rem] h-[35rem] ${currentScreen.includes('2') ? 'bg-purple-100/50' : 'bg-orange-100/50'} rounded-full mix-blend-multiply filter blur-[120px] opacity-70 transition-colors duration-1000`}></div>
      
      <style dangerouslySetInnerHTML={{__html: `@keyframes fadeInUp { from { opacity: 0; transform: translateY(20px); } to { opacity: 1; transform: translateY(0); } } .animate-fade-in { animation: fadeInUp 0.6s cubic-bezier(0.16, 1, 0.3, 1) forwards; opacity: 0; }`}} />

      <header className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${isScrolled ? 'bg-[#FCFAFA]/90 backdrop-blur-md border-b border-gray-100 shadow-sm' : 'bg-transparent'}`}>
        <div className="max-w-5xl mx-auto px-6 h-16 flex items-center justify-between">
          <div onClick={() => setCurrentScreen('landing')} className="font-semibold tracking-tight text-lg text-[#4A3D3F] cursor-pointer hover:text-[#D37B85] transition-colors">{t.navTitle}</div>
          {(!currentScreen.includes('test') && !currentScreen.includes('result')) && (
            <div className="flex bg-white/60 p-1 rounded-full border border-gray-100 backdrop-blur-sm">
              <button onClick={() => setLang('ru')} className={`px-4 py-1 text-xs font-medium rounded-full transition-all duration-300 ${lang === 'ru' ? 'bg-white shadow-sm text-black' : 'text-gray-500 hover:text-gray-800'}`}>РУС</button>
              <button onClick={() => setLang('kz')} className={`px-4 py-1 text-xs font-medium rounded-full transition-all duration-300 ${lang === 'kz' ? 'bg-white shadow-sm text-black' : 'text-gray-500 hover:text-gray-800'}`}>ҚАЗ</button>
            </div>
          )}
        </div>
      </header>

      <main className="relative pt-32 px-6 max-w-5xl mx-auto z-10">
        {currentScreen === 'landing' && renderLanding()}
        {currentScreen === 'registration' && renderRegistration()}
        {currentScreen === 'instructions' && renderInstructions()}
        
        {/* УРОВЕНЬ 1 */}
        {currentScreen === 'test1' && renderTestGeneric(
          questionsList, currentPage1, setCurrentPage1, answers1, handleAnswer1, canGoNext1, 
          () => calculateResults1(answers1), 
          () => { const dummy = questionsList.map(() => Math.random() < 0.4 ? 'yes' : Math.random() < 0.8 ? 'no' : 'maybe'); setAnswers1(dummy); calculateResults1(dummy); },
          [{ val: 'yes', label: t.btnYes }, { val: 'maybe', label: t.btnMaybe }, { val: 'no', label: t.btnNo }]
        )}
        {currentScreen === 'result1' && renderResult1()}

        {/* УРОВЕНЬ 2 */}
        {currentScreen === 'level2_intro' && renderLevel2Intro()}
        {currentScreen === 'test2' && renderTestGeneric(
          questionsLevel2, currentPage2, setCurrentPage2, answers2, handleAnswer2, canGoNext2, 
          () => generateReport2(answers2),
          () => { const dummy = questionsLevel2.map(() => Math.random() < 0.5 ? 'yes' : 'no'); setAnswers2(dummy); generateReport2(dummy); },
          [{ val: 'yes', label: t.btnYes }, { val: 'no', label: t.btnNo }]
        )}
        {currentScreen === 'result2' && renderResult2()}
      </main>
    </div>
  );
}
